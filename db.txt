// ================================================================
// CGV CINEMA ENTERPRISE — DATABASE SCHEMA (DBML for dbdiagram.io)
// Chuan hoa 100% theo cac JPA Entity thuc te trong Backend Microservices
// Ngay cap nhat: 2026-09-25
// ================================================================

// Luu y kien truc Database-per-Service:
// - He thong gom 5 Database rieng biet tuong ung 5 Microservices co Entity:
//   1. IdentityService (Quan ly nguoi dung, hang thanh vien, OAuth2)
//   2. CatalogService  (Phim, the loai, dien vien, cum rap, phong chieu, ghe, lich chieu, tin tuc, su kien)
//   3. BookingService  (Giu ghe, don dat ve, chi tiet ghe dat, outbox su kien)
//   4. PaymentService  (Giao dich thanh toan VNPay, nhat ky payment logs, outbox payment)
//   5. MarketingService(Chien dich khuyen mai, ma giam gia, lich su su dung voucher)
// - 2 Services khong su dung Database (Stateless Services):
//   + NotificationService: Lang nghe Kafka topic 'notification.send' de gui email OTP qua Brevo REST API.
//   + MediaService: Xu ly upload poster/banner truc tiep len AWS S3 Bucket.

// ================================================================
// 1. IDENTITY SERVICE
// ================================================================

TableGroup IdentityService {
  membership_tiers
  users
  user_connected_accounts
}

Table membership_tiers {
  code varchar [pk, note: 'Ma hang hoi vien: MEMBER, VIP, VVIP']
  name varchar [not null, note: 'Ten hang: Thanh vien Thuong, VIP, VVIP']
  min_spend decimal [not null, default: 0, note: 'Chi tieu toi thieu trong nam de thiet lap hang']
  description text [note: 'Mo ta dac quyen va quyen loi hang']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
}

Table users {
  id varchar [pk, note: 'ID nguoi dung tu Keycloak (UUID String). Khong luu password/role o day']
  email varchar [unique, not null]
  full_name varchar [note: 'Ho va ten day du']
  phone varchar(20) [note: 'So dien thoai lien he']
  membership_tier varchar [not null, default: 'MEMBER', ref: > membership_tiers.code, note: 'Khoa ngoai toi hang thanh vien']
  total_spend_ytd decimal [not null, default: 0, note: 'Tong chi tieu tich luy trong nam de xet len hang']
  version bigint [not null, default: 0, note: 'OPTIMISTIC LOCKING (@Version): Chong ghi de mat diem khi nhieu luong cong chi tieu dong thoi']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    email [name: 'idx_cgv_user_email']
    membership_tier [name: 'idx_cgv_user_membership_tier']
    phone [name: 'idx_cgv_user_phone']
  }
}

Table user_connected_accounts {
  id uuid [pk]
  user_id varchar [not null, ref: > users.id, note: 'Tham chieu nguoi dung']
  provider varchar [not null, note: 'GOOGLE, FACEBOOK, APPLE']
  provider_id varchar [not null, note: 'ID nguoi dung tren nha cung cap OAuth']
  provider_email varchar [note: 'Email tra ve tu nha cung cap']
  access_token text [note: 'OAuth access token']
  token_expires_at timestamp [note: 'Thoi han het han cua token']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    (provider, provider_id) [unique, name: 'uk_cgv_provider_provider_id']
    user_id [name: 'idx_cgv_connected_user_id']
  }
}

// ================================================================
// 2. CATALOG SERVICE
// ================================================================

TableGroup CatalogService {
  regions
  cinemas
  cinema_amenities
  rooms
  seat_types
  seats
  genres
  movies
  movie_genres
  movie_casts
  showtimes
  articles
  events
}

Table regions {
  id integer [pk, increment]
  name varchar [unique, not null, note: 'Ten khu vuc: Ho Chi Minh, Ha Noi, Da Nang...']
  slug varchar [unique, not null, note: 'Slug URL: ho-chi-minh, ha-noi']
}

Table cinemas {
  id uuid [pk]
  region_id integer [not null, ref: > regions.id]
  name varchar [unique, not null, note: 'Ten cum rap: CGV Hung Vuong Plaza, CGV Landmark 81']
  address varchar [note: 'Dia chi chi tiet cua cum rap']
  phone varchar(20) [note: 'Hotline lien he']
  opening_hours varchar [note: 'Gio mo cua: 08:00 - 24:00']
  latitude double [note: 'Vi do toa do ban do']
  longitude double [note: 'Kinh do toa do ban do']
  status varchar [not null, default: 'ACTIVE', note: 'CinemaStatus: ACTIVE, INACTIVE, MAINTENANCE']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    region_id [name: 'idx_cinema_region_id']
    status [name: 'idx_cinema_status']
  }
}

Table cinema_amenities {
  cinema_id uuid [not null, ref: > cinemas.id]
  amenity varchar [not null, note: 'CinemaAmenityType: PARKING, POPCORN_BAR, IMAX, 4DX, GOLD_CLASS, SWEETBOX, WHEELCHAIR_ACCESS, ATM']

  indexes {
    (cinema_id, amenity) [pk]
    amenity [name: 'idx_cinema_amenity']
  }
}

Table rooms {
  id uuid [pk]
  cinema_id uuid [not null, ref: > cinemas.id]
  name varchar [not null, note: 'Ten phong: Cinema 1, IMAX Hall, Gold Class 2']
  total_seats integer [default: 0, note: 'Tong so luong ghe']
  row_count integer [not null, note: 'So luong hang ghe (A, B, C...)']
  column_count integer [not null, note: 'So luong cot ghe moi hang (1, 2, 3...)']
  room_type varchar [not null, default: 'STANDARD', note: 'RoomType: STANDARD, IMAX, 4DX, GOLD_CLASS, SCREENX']
  status varchar [not null, default: 'ACTIVE', note: 'RoomStatus: ACTIVE, MAINTENANCE, INACTIVE']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    (cinema_id, name) [unique, name: 'uk_rooms_cinema_id_name']
  }
}

Table seat_types {
  name varchar [pk, note: 'SeatTypeName: STANDARD, VIP, SWEETBOX']
  surcharge decimal [not null, default: 0, note: 'Phu thu them tren gia ve co ban: Standard=0, VIP=15000, Sweetbox=30000']
  description text [note: 'Mo ta kieu ghe']
}

Table seats {
  id uuid [pk]
  room_id uuid [not null, ref: > rooms.id]
  row_char varchar(2) [not null, note: 'Ky tu hang: A, B, C, ..., H']
  seat_number integer [not null, note: 'So thu tu ghe: 1, 2, 3...']
  seat_type_name varchar [not null, ref: > seat_types.name]
  is_active boolean [default: true, note: 'false neu ghe dang bi hong hoc/bao tri tai rap']

  indexes {
    (room_id, row_char, seat_number) [unique, name: 'idx_seats_room_id_row_char_seat_number']
    room_id [name: 'idx_seats_room_id']
  }
}

Table genres {
  id integer [pk, increment]
  name varchar [unique, not null, note: 'Hanh dong, Hai huoc, Kinh di, Khoa hoc vien tuong']
  slug varchar [unique, not null, note: 'hanh-dong, hai-huoc, kinh-di']
}

Table movies {
  id uuid [pk]
  title varchar [unique, not null, note: 'Ten phim tieng Viet']
  original_title varchar [note: 'Ten phim goc quoc te']
  synopsis text [note: 'Noi dung tom tat phim']
  director varchar [note: 'Dao dien']
  language varchar [note: 'Tieng Viet, Tieng Anh, Tieng Han']
  subtitle varchar [note: 'Phu de tieng Viet, Tieng Anh']
  supported_modes varchar [note: 'Cac dinh dang ho tro: 2D, 3D, IMAX']
  age_rating varchar(5) [note: 'P (moi lua tuoi), K (duoi 13 tuoi kem PH), T13, T16, T18, C (cam)']
  duration_minutes integer [note: 'Thoi luong phim (phut)']
  release_date date [note: 'Ngay khoi chieu']
  end_date date [note: 'Ngay ket thuc chieu du kien']
  showing_status varchar [not null, note: 'MovieShowingStatus: COMING_SOON, NOW_SHOWING, END_SHOWING']
  poster_url varchar [note: 'Duong dan anh poster tren S3']
  backdrop_url varchar [note: 'Duong dan anh bia backdrop tren S3']
  trailer_youtube_url varchar [note: 'Link trailer YouTube']
  is_featured boolean [not null, default: false, note: 'Phim noi bat tren trang chu']
  status varchar [not null, default: 'ACTIVE', note: 'MovieStatus: ACTIVE, INACTIVE, DRAFT']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    showing_status [name: 'idx_movies_showing_status']
    release_date [name: 'idx_movies_release_date']
    status [name: 'idx_movies_status']
  }
}

Table movie_genres {
  movie_id uuid [not null, ref: > movies.id]
  genre_id integer [not null, ref: > genres.id]

  indexes {
    (movie_id, genre_id) [pk]
    genre_id [name: 'idx_movie_genres_genre_id']
    movie_id [name: 'idx_movie_genres_movie_id']
  }
}

Table movie_casts {
  id uuid [pk]
  movie_id uuid [not null, ref: > movies.id]
  actor_name varchar [not null, note: 'Ten dien vien / dao dien']
  character_name varchar [note: 'Ten nhan vat trong phim']
  role_type varchar [note: 'CastRoleType: ACTOR, DIRECTOR, PRODUCER']
  avatar_url varchar [note: 'Anh chan dung dien vien']
  display_order integer [default: 0]

  indexes {
    movie_id [name: 'idx_movie_casts_movie_id']
  }
}

Table showtimes {
  id uuid [pk]
  movie_id uuid [not null, ref: > movies.id]
  room_id uuid [not null, ref: > rooms.id]
  show_date date [not null, note: 'Ngay chieu (YYYY-MM-DD)']
  start_time timestamp [not null, note: 'Gio bat dau chieu']
  end_time timestamp [not null, note: 'Gio ket thuc chieu']
  language varchar [note: 'Ngon ngu am thanh']
  subtitle_language varchar [note: 'Ngon ngu phu de']
  viewing_mode varchar [note: 'ViewingMode: _2D, _3D, IMAX, _4DX']
  format varchar [note: 'ShowtimeFormat: DUBBED (Long tieng), SUBTITLED (Phu de), ORIGINAL']
  base_price decimal [not null, note: 'Gia ve co ban cho suat chieu nay (chua tinh phu thu ghe)']
  available_seats integer [note: 'So ghe con trong']
  status varchar [not null, default: 'ACTIVE', note: 'ShowtimeStatus: ACTIVE, CANCELLED, COMPLETED']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    movie_id [name: 'idx_showtimes_movie_id']
    room_id [name: 'idx_showtimes_room_id']
    show_date [name: 'idx_showtimes_show_date']
    status [name: 'idx_showtimes_status']
    start_time [name: 'idx_showtimes_start_time']
  }
}

Table articles {
  id uuid [pk]
  title varchar [not null, note: 'Tieu de bai viet / tin tuc / blog dien anh']
  slug varchar [unique, not null, note: 'Duong dan SEO-friendly']
  thumbnail_url varchar [note: 'Anh dai dien bai viet']
  content text [not null, note: 'Noi dung chi tiet (HTML/Markdown)']
  excerpt text [note: 'Doan trich dan ngan']
  category varchar [not null, note: 'ArticleCategory: NEWS, PROMOTION, REVIEW, EVENT']
  status varchar [not null, default: 'DRAFT', note: 'ArticleStatus: DRAFT, PUBLISHED, ARCHIVED']
  tags varchar [note: 'Cac the tag cach nhau boi dau phay']
  author_name varchar [note: 'Tac gia bai viet']
  views integer [default: 0, note: 'Luot xem']
  is_featured boolean [default: false, note: 'Tin noi bat']
  is_trending boolean [default: false, note: 'Tin thinh hanh xu huong']
  movie_id uuid [ref: > movies.id, note: 'Lien ket voi phim neu la bai review phim']
  published_at timestamp [note: 'Thoi diem xuat ban']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    slug [unique, name: 'idx_articles_slug']
    status [name: 'idx_articles_status']
    category [name: 'idx_articles_category']
    published_at [name: 'idx_articles_published_at']
    is_featured [name: 'idx_articles_is_featured']
    is_trending [name: 'idx_articles_is_trending']
  }
}

Table events {
  id uuid [pk]
  title varchar [not null, note: 'Tieu de su kien: Premiere, Giao luu fan, Tri an hoi vien']
  description text [not null, note: 'Noi dung chi tiet su kien']
  thumbnail_url varchar [note: 'Anh banner su kien']
  event_date date [not null, note: 'Ngay dien ra su kien']
  event_time time [not null, note: 'Gio dien ra su kien']
  location_name varchar [not null, note: 'Ten dia diem / ranh gioi to chuc']
  cinema_id uuid [ref: > cinemas.id, note: 'Cum rap dien ra su kien (neu co)']
  registration_url varchar [note: 'Link form dang ky tham gia']
  max_attendees integer [note: 'So luong nguoi tham gia toi da']
  current_attendees integer [default: 0, note: 'So luong da dang ky hien tai']
  status varchar [not null, default: 'UPCOMING', note: 'EventStatus: UPCOMING, ONGOING, COMPLETED, CANCELLED']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    event_date [name: 'idx_events_event_date']
    status [name: 'idx_events_status']
    cinema_id [name: 'idx_events_cinema_id']
  }
}

TableGroup BookingService {
  bookings
  booking_seats
  outbox_booking_events
}

Table bookings {
  id uuid [pk]
  user_id varchar [note: 'ID nguoi dung tu IdentityService hoac guest_phone neu la khach vang lai']
  guest_name varchar(100) [note: 'Ho ten khach vang lai']
  guest_email varchar(100) [note: 'Email khach vang lai nhan thong tin ve']
  guest_phone varchar(20) [note: 'SDT khach vang lai']
  showtime_id uuid [not null, note: 'Tham chieu logic toi CatalogService.showtimes.id']
  promotion_id uuid [note: 'Tham chieu logic toi MarketingService.promotions.id (neu co dung voucher)']
  total_base_amount decimal [not null, note: 'Tong tien ve goc (chua tru voucher)']
  discount_amount decimal [default: 0, note: 'So tien duoc giam tu voucher']
  final_amount decimal [not null, note: 'So tien thuc te khach phai thanh toan qua VNPay']
  payment_deadline timestamp [note: 'Han chot thanh toan (Mac dinh 10 phut sau khi tao don)']
  qr_code_url varchar [note: 'URL ma QR de render ra ma soat ve tai rap']
  status varchar [not null, default: 'PAYMENT_PENDING', note: 'BookingStatus: PAYMENT_PENDING, CONFIRMED, CANCELLED, REFUNDED, CHECKED_IN']
  cancelled_at timestamp [note: 'Thoi diem don bi huy neu qua han thanh toan']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    user_id [name: 'idx_cgv_booking_user_id']
    showtime_id [name: 'idx_cgv_booking_showtime_id']
    status [name: 'idx_cgv_booking_status']
    created_at [name: 'idx_cgv_booking_created_at']
  }
}

Table booking_seats {
  id uuid [pk]
  booking_id uuid [not null, ref: > bookings.id, note: 'Don dat ve so huu ghe nay']
  seat_id uuid [not null, note: 'Tham chieu logic toi CatalogService.seats.id']
  showtime_id uuid [not null, note: 'Tham chieu logic toi CatalogService.showtimes.id']
  price decimal [not null, note: 'Gia ban cua ghe nay (gia goc suat chieu + phu thu loai ghe)']
  seat_label varchar [note: 'Nhan ghe hien thi truc quan: A01, H05, VIP-C02']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    (seat_id, booking_id) [name: 'idx_cgv_booking_seat_booking_id_seat_id']
    (showtime_id, seat_id) [name: 'idx_cgv_booking_seat_showtime_id_seat_id']
  }
}


Table outbox_booking_events {
  id uuid [pk]
  aggregate_id varchar [not null, note: 'ID doi tuong nghiep vu (Booking ID)']
  aggregate_type varchar [not null, note: 'Loai doi tuong: BOOKING']
  eventType varchar [not null, note: 'Ten su kien: BookingCreatedEvent, BookingConfirmedEvent']
  payload text [not null, note: 'Chuoi JSON du lieu su kien']
  is_published boolean [default: false, note: 'Trang thai da day sang Kafka broker hay chua']
  retry_count integer [default: 0, note: 'So lan retry day event neu Kafka gap su co']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    (is_published, created_at) [name: 'idx_cgv_outbox_created_at_is_published']
    aggregate_id [name: 'idx_cgv_outbox_aggregate_id']
  }
}

// 4. PAYMENT SERVICE

TableGroup PaymentService {
  payments
  payment_logs
  outbox_payment_events
}

Table payments {
  id uuid [pk]
  provider varchar [not null, note: 'Cong thanh toan: VNPAY, MOMO, ZALOPAY']
  booking_id uuid [not null, note: 'Tham chieu logic toi BookingService.bookings.id']
  user_id varchar [not null, note: 'ID nguoi dung thuc hien thanh toan']
  amount decimal [not null, note: 'So tien giao dich']
  status varchar [not null, default: 'PENDING', note: 'PaymentStatus: PENDING, SUCCESS, FAILED, REFUNDED']
  transaction_id varchar [note: 'Ma giao dich phia cong thanh toan tra ve (vnp_TxnRef, vnp_TransactionNo)']
  payment_method varchar [note: 'Phuong thuc: VNPAY_NCB, VNPAY_VISA, VNPAY_QR...']
  currency varchar [default: 'VND']
  paid_at timestamp [note: 'Thoi diem thanh toan thanh cong']
  refund_amount decimal [note: 'So tien hoan lai neu giao dich bi huy']
  refunded_at timestamp [note: 'Thoi diem hoan tien']
  raw text [note: 'Toan bo thong tin payload phan hoi nguyen ban tu cong thanh toan']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    booking_id [name: 'idx_cgv_payment_booking_id']
    status [name: 'idx_cgv_payment_status']
    transaction_id [name: 'idx_cgv_payment_transaction_id']
    provider [name: 'idx_cgv_payment_provider']
  }
}

Table payment_logs {
  id uuid [pk]
  payment_id uuid [not null, ref: > payments.id, note: 'Giao dich thanh toan can ghi log']
  description text [note: 'Mo ta chi tiet buoc xu ly / trang thai loi']
  raw_request text [note: 'Request gui sang cong thanh toan']
  raw_response text [note: 'Response phan hoi tu cong thanh toan']
  ip_address varchar [note: 'Dia chi IP cua client thuc hien request']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    payment_id [name: 'idx_cgv_payment_log_payment_id']
    created_at [name: 'idx_cgv_payment_created_at']
  }
}

Table outbox_payment_events {
  id uuid [pk]
  aggregate_id varchar [not null, note: 'ID doi tuong nghiep vu (Payment ID)']
  aggregate_type varchar [not null, note: 'Loai doi tuong: PAYMENT']
  eventType varchar [not null, note: 'Ten su kien: PaymentCompletedEvent, PaymentFailedEvent']
  payload text [not null, note: 'Chuoi JSON du lieu su kien']
  is_published boolean [default: false, note: 'Trang thai da day sang Kafka topic payment.completed']
  retry_count integer [default: 0, note: 'So lan thu lai neu Kafka broker bi ngat ket noi']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    (is_published, created_at) [name: 'idx_cgv_outbox_created_at_is_published']
    aggregate_id [name: 'idx_cgv_outbox_aggregate_id']
  }
}

// ================================================================
// 5. MARKETING SERVICE
// ================================================================

TableGroup MarketingService {
  promotions
  promotion_usages
}

Table promotions {
  id uuid [pk]
  code varchar(50) [unique, not null, note: 'Ma voucher giam gia: CGV2026, VIPDISCOUNT, TIKTOK50K']
  name varchar(200) [not null, note: 'Ten chuong trinh uu dai']
  description text [note: 'Noi dung chi tiet the le chuong trinh']
  discount_type varchar(20) [not null, note: 'DiscountType: PERCENTAGE (giam %), FIXED_AMOUNT (giam so tien co dinh)']
  discount_value decimal [not null, note: 'Gia tri giam: VD 20 cho 20%, hoac 50000 cho 50.000d']
  max_discount_amount decimal [note: 'So tien giam toi da khi chon kieu PERCENTAGE']
  min_order_value decimal [default: 0, note: 'Gia tri don hang toi thieu de duoc ap dung voucher']
  applicable_tier varchar(20) [note: 'Hang thanh vien toi thieu: ALL, MEMBER, VIP, VVIP']
  valid_from timestamp [not null, note: 'Ngay bat dau co hieu luc']
  valid_to timestamp [not null, note: 'Ngay het han ap dung']
  usage_limit integer [note: 'Tong so luong ma phat hanh (null neu khong gioi han)']
  max_uses_per_user integer [not null, default: 1, note: 'So lan toi da 1 user duoc phep su dung ma nay']
  is_active boolean [not null, default: true, note: 'Trang thai voucher con hoat dong hay khong']
  version bigint [not null, note: 'OPTIMISTIC LOCKING (@Version): Chong vuot han muc usage_limit khi hang ngan nguoi cung dung voucher']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    (is_active, valid_from, valid_to) [name: 'idx_promotions_active_dates']
    (applicable_tier, min_order_value) [name: 'idx_promotions_tier_min_order']
  }
}

Table promotion_usages {
  id uuid [pk]
  promotion_id uuid [not null, ref: > promotions.id, note: 'Ma voucher da su dung']
  user_id varchar [not null, note: 'Nguoi dung da su dung voucher']
  booking_id uuid [unique, not null, note: 'Don dat ve ap dung voucher (Moi don chi dung duy nhat 1 voucher)']
  discount_applied decimal(12,2) [not null, note: 'So tien giam thuc te da duoc tru vao don']
  used_at timestamp [not null, default: `now()`, note: 'Thoi diem su dung ma']

  indexes {
    (promotion_id, user_id) [name: 'idx_promo_usage_user']
    booking_id [unique, name: 'idx_promo_usage_booking']
  }
}
